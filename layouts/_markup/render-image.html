{{- /* Images in a page bundle are served as WebP, at most 1600 px wide; the original stays in the repo. Anything else (remote URLs, SVG, GIF) is passed through. */ -}} {{- $u := urls.Parse .Destination -}} {{- $img := "" -}} {{- if not $u.IsAbs }}{{ $img = .Page.Resources.Get $u.Path }}{{ end -}} {{- if and $img (in (slice "png" "jpeg") $img.MediaType.SubType) -}} {{- $out := $img.Process "webp q82" -}} {{- if gt $img.Width 1600 }}{{ $out = $img.Resize "1600x webp q82" }}{{ end -}} {{- else -}} {{- end -}}
